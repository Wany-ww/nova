[INIT 4000x3000]

// open 4000x3000 YUV8
open 3000, 8000, 12
set_test_cam 0
set_mipi_test_cam 0

// MIPI & Frame 
set_mipi_default 0x1e
//set_mipi_input 4, 0
//set_mipi_info 0, 0x1e, 0, 0, 0
//set_mipi_dsi 0, 0, 0, 0

// POWER on
set_ism_io 0 1 1
//set_cam_power 0
// ISM Volt&Freq
set_ism_default
//set_ism_volt 1.8
//set_ism_freq 0, 24.0

// ISM Reset
delay 100
set_ism_reset 1
delay 5
set_ism_reset 0
delay 200

// I2C prepare
set_i2c_freq 200.0

// Slave Address

set_i2c 1 1 0x78 0xF0
delay 1000

# Set Resolution
# 4000 x 3000
set_i2c 1 1 0x78 0x54 46 0x00

# Set Preview rate
set_i2c 1 1 0x78 0x5A 60 0

#preview start
#set_i2c 1 1 0x78 0x6B 0x01
#live view start
set_i2c 1 1 0x78 0x65 0x01
